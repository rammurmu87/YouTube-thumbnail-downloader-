<script>

  const youtubeInput = document.getElementById("youtubeUrl");
  const getThumbnailBtn = document.getElementById("getThumbnailBtn");
  const downloadBtn = document.getElementById("downloadBtn");
  const openBtn = document.getElementById("openBtn");

  const thumbnail = document.getElementById("thumbnail");
  const result = document.getElementById("result");
  const error = document.getElementById("error");

  const loader = document.getElementById("loader");
  const buttonText = document.getElementById("buttonText");

  let currentThumbnailUrl = "";
  let currentVideoId = "";


  // Extract YouTube Video ID
  function getYouTubeVideoId(url) {

    try {

      const parsedUrl = new URL(url);

      // youtube.com/watch?v=VIDEO_ID
      if (
        parsedUrl.hostname.includes("youtube.com") &&
        parsedUrl.searchParams.get("v")
      ) {
        return parsedUrl.searchParams.get("v");
      }

      // youtu.be/VIDEO_ID
      if (parsedUrl.hostname === "youtu.be") {
        return parsedUrl.pathname.substring(1).split("/")[0];
      }

      // youtube.com/shorts/VIDEO_ID
      if (parsedUrl.pathname.startsWith("/shorts/")) {
        return parsedUrl.pathname
          .split("/shorts/")[1]
          .split("/")[0];
      }

      // youtube.com/embed/VIDEO_ID
      if (parsedUrl.pathname.startsWith("/embed/")) {
        return parsedUrl.pathname
          .split("/embed/")[1]
          .split("/")[0];
      }

      return null;

    } catch (e) {

      return null;

    }

  }


  // Show error
  function showError(message) {

    error.textContent = message;
    error.style.display = "block";
    result.style.display = "none";

  }


  // Hide error
  function hideError() {

    error.style.display = "none";

  }


  // Get Thumbnail
  async function getThumbnail() {

    const url = youtubeInput.value.trim();

    hideError();

    if (!url) {

      showError("Please paste a YouTube video URL.");
      return;

    }

    const videoId = getYouTubeVideoId(url);

    if (!videoId) {

      showError(
        "Invalid YouTube URL. Please paste a valid YouTube video link."
      );

      return;

    }


    // Loading
    buttonText.style.display = "none";
    loader.style.display = "inline-block";
    getThumbnailBtn.disabled = true;


    currentVideoId = videoId;

    // Max resolution thumbnail
    const maxRes =
      `https://img.youtube.com/vi/${videoId}/maxresdefault.jpg`;

    const highQuality =
      `https://img.youtube.com/vi/${videoId}/hqdefault.jpg`;


    // Check maxres thumbnail
    const img = new Image();

    img.onload = function() {

      currentThumbnailUrl = maxRes;

      thumbnail.src = maxRes;

      result.style.display = "block";

      resetButton();

    };


    img.onerror = function() {

      // Fallback to HQ thumbnail
      currentThumbnailUrl = highQuality;

      thumbnail.src = highQuality;

      result.style.display = "block";

      resetButton();

    };


    img.src = maxRes;

  }


  function resetButton() {

    buttonText.style.display = "inline";
    loader.style.display = "none";
    getThumbnailBtn.disabled = false;

  }


  // Download thumbnail
  async function downloadThumbnail() {

    if (!currentThumbnailUrl) {
      return;
    }

    try {

      const response =
        await fetch(currentThumbnailUrl);

      if (!response.ok) {
        throw new Error("Download failed");
      }

      const blob =
        await response.blob();

      const blobUrl =
        URL.createObjectURL(blob);

      const link =
        document.createElement("a");

      link.href = blobUrl;

      link.download =
        `youtube-thumbnail-${currentVideoId}.jpg`;

      document.body.appendChild(link);

      link.click();

      document.body.removeChild(link);

      URL.revokeObjectURL(blobUrl);

    } catch (err) {

      // Fallback: open image
      window.open(
        currentThumbnailUrl,
        "_blank"
      );

    }

  }


  // Open full size
  function openFullSize() {

    if (!currentThumbnailUrl) {
      return;
    }

    window.open(
      currentThumbnailUrl,
      "_blank"
    );

  }


  // Button events
  getThumbnailBtn.addEventListener(
    "click",
    getThumbnail
  );

  downloadBtn.addEventListener(
    "click",
    downloadThumbnail
  );

  openBtn.addEventListener(
    "click",
    openFullSize
  );


  // Press Enter
  youtubeInput.addEventListener(
    "keydown",
    function(event) {

      if (event.key === "Enter") {
        getThumbnail();
      }

    }
  );


</script>
